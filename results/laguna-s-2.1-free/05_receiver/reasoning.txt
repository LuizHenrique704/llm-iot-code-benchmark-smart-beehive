#include <Wire.h>
#include <LoRa_E32.h>
#include <HardwareSerial.h>
#include "esp_sleep.h"

#define UART_RX 16
#define UART_TX 17
#define AUX_PIN 5
#define M0_PIN 18
#define M1_PIN 19

HardwareSerial uart2(2);
LoRa_E32 e32ttl(&uart2, AUX_PIN, M0_PIN, M1_PIN);

typedef struct __attribute__((packed)) {
    char identifier[8];
    uint8_t mac[6];
    float temperature;
    float humidity;
    float pressure;
    float altitude;
    uint8_t hiveState;
    uint32_t transmissions;
} TelemetryData;

TelemetryData latestData;

void setup() {
    Serial.begin(115200);
    delay(100);

    uart2.begin(9600, SERIAL_8N1, UART_RX, UART_TX);
    delay(100);

    e32ttl.setMode(MODE_CONFIG);
    e32ttl.setAddH(0);
    e32ttl.setAddL(3);
    e32ttl.setChan(13);
    e32ttl.setUARTbaud(UART_BPS_9600);
    e32ttl.setUARTparity(MODE_00_8N1);
    e32ttl.setAirdatarate(AIR_DATA_RATE_010_24);
    e32ttl.setWakeup(WAKE_UP_250);
    e32ttl.setIO_Mode(IO_D_MODE_PUSH_PULLS_PULL_UPS);
    e32ttl.setFEC(FEC_0_OFF);
    e32ttl.setPower(POWER_20);
    e32ttl.setFixedTransmission(FT_FIXED_TRANSMISSION);
    e32ttl.saveParams();
    e32ttl.setMode(MODE_NORMAL);
}

void loop() {
    uint8_t buffer[sizeof(TelemetryData)];
    int result = e32ttl.receiveFixedMessage(0, 1, 13, buffer, sizeof(buffer));

    if (result > 0) {
        TelemetryData* data = (TelemetryData*)buffer;

        if (strcmp(data->identifier, "colmeia") == 0) {
            latestData = *data;

            // Print MAC
            Serial.print("MAC: ");
            for (int i = 0; i < 6; i++) {
                if (i > 0) Serial.print(":");
                Serial.print(latestData.mac[i], HEX);
            }
            Serial.println();

            // Print temperature
            Serial.print("Temperatura: ");
            Serial.print(latestData.temperature, 2);
            Serial.println(" C");

            // Print humidity
            Serial.print("Umidade: ");
            Serial.print(latestData.humidity, 2);
            Serial.println(" %");

            // Print pressure
            Serial.print("Pressao: ");
            Serial.print(latestData.pressure, 2);
            Serial.println(" hPa");

            // Print altitude
            Serial.print("Altitude: ");
            Serial.print(latestData.altitude, 2);
            Serial.println(" m");

            // Print hive state
            Serial.print("Estado: ");
            if (latestData.hiveState == 1) {
                Serial.println("ABERTA");
            } else {
                Serial.println("FECHADA");
            }

            // Print transmission count
            Serial.print("Contador: ");
            Serial.println(latestData.transmissions);
        }
    } else if (result < 0) {
        Serial.println("Erro na recepcao do pacote.");
    }
}
